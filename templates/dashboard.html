SELECT c.course_name AS Course, AVG(g.marks) AS Average_Marks
FROM Grades g
JOIN Enrollment e ON g.enroll_id = e.enroll_id
JOIN Courses c ON e.course_id = c.course_id
GROUP BY c.course_name;
